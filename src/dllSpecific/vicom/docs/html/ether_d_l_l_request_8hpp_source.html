60// ----------------------------------------------------------------------
61// Get parameter value from JSON with default fallback
62// @brief Template function to safely retrieve parameters from JSON
63// @tparam T Type of the parameter value
64// @param reqArguments JSON object containing the parameters
65// @param key Parameter key to look for
66// @param defaultValue Default value if key not found
67// @return Value from JSON or default if key doesn't exist
68// @note Logs debug message when using default value
69// @ingroup vicom_request
70// ----------------------------------------------------------------------
71template<typename T>
72T getParamValue(const json& reqArguments, const char* key, T defaultValue) {
73 if (reqArguments.contains(key)) {
74 return reqArguments[key].get<T>();
75 }
76 loggerPtr->debug("Using default value for parameter: {}", key);
77 return defaultValue;
78}

417// ----------------------------------------------------------------------
418// Request queue processor (runs in separate thread)
419// @brief Thread-safe queue consumer that processes incoming requests
420// @param DLLConn Connection data for the ViCom DLL
421// @param request Message queue for incoming requests
422// @param response Message queue for outgoing responses
423// @param interruptionCode Signal for service interruption
424// @param refMeasurementInProgress Flag to track measurement state
425// @note This function blocks until interruption signal is received
426// @note Must be run in a dedicated thread
427// @ingroup vicom_request
428// ----------------------------------------------------------------------
429void processRequestQueue(
430 DLLConnectionData& DLLConn,
431 MessageQueue& request,
432 MessageQueue& response,
433 const std::atomic<edll::INT_CODE>& interruptionCode,
434 std::atomic<bool>& refMeasurementInProgress
435)
436{
437 const std::string funcName = "processRequestQueue";
438
439 while (interruptionCode == edll::Code::RUNNING)
440 {
441 loggerPtr->debug("processRequestQueue: waiting for request");
442 json oneRequest = request.waitAndPop(interruptionCode, funcName);
443 loggerPtr->debug("processRequestQueue: got request, processing...");
444
445 if (interruptionCode != edll::Code::RUNNING) {
446 loggerPtr->debug("processRequestQueue: interruption detected, breaking");
447 break;
448 }
449
450 unsigned long cmd = oneRequest.value(TaskKeys::CommandCode::VALUE, TaskKeys::CommandCode::INIT_VALUE);
451 loggerPtr->debug("processRequestQueue: cmd={}", cmd);
452
453 if (!validRequest(oneRequest, cmd, response)) {
454 loggerPtr->debug("processRequestQueue: invalid request, continuing");
455 continue;
456 }
457
458 loggerPtr->debug("processRequestQueue: calling DLLFunctionCall");
459 DLLFunctionCall(DLLConn, oneRequest, cmd, refMeasurementInProgress, interruptionCode);
460 loggerPtr->debug("processRequestQueue: DLLFunctionCall returned");
461 }
462 loggerPtr->debug("processRequestQueue: exiting loop");
463}
