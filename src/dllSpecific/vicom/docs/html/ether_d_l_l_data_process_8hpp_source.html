59// ----------------------------------------------------------------------
60// Process Power Scan measurement result
61// @brief Converts Power Scan measurement data to JSON format
62// @param sweepSettings Pointer to sweep settings used in measurement
63// @param pResult Pointer to the measurement result structure
64// @return JSON object containing count, frequencies, and spectrum data
65// @note Spectrum values are encoded as Base64 string
66// @ingroup vicom_processing
67// ----------------------------------------------------------------------
68inline json processPowerScanResult(const SSweepSettings* sweepSettings, const RFPOWERSCAN::SMeasResult* pResult)
69{
70 using namespace RohdeSchwarz::ViCom::RFPOWERSCAN;
71
72 json resultJson;
73
74 if (!pResult || !pResult->pSpectrumResult) {
75 resultJson["error"] = "Invalid or null result pointer.";
76 return resultJson;
77 }
78
79 const RFPOWERSCAN::SMeasResult::SSpectrumResult* spectrumResult = pResult->pSpectrumResult;
80 auto startFreq = sweepSettings->dStartFrequencyInHz;
81 auto stopFreq = sweepSettings->dStopFrequencyInHz;
82
83 resultJson["count"] = spectrumResult->dwCount;
84 resultJson["startFrequency"] = startFreq;
85 resultJson["stopFrequency"] = stopFreq;
86
87 // Convert the float array of spectrum values to a Base64 encoded string
88 if (spectrumResult->dwCount > 0 && spectrumResult->pfSpectrumValuesInDBm)
89 {
90 const unsigned char* dataPtr = reinterpret_cast<const unsigned char*>(spectrumResult->pfSpectrumValuesInDBm);
91 size_t dataSize = spectrumResult->dwCount * sizeof(float);
92 resultJson["spectrum_dbm_base64"] = base64Encode(dataPtr, static_cast<unsigned int>(dataSize));
93 }
94 else
95 {
96 resultJson["spectrum_dbm_base64"] = "";
97 }
98
99 return resultJson;
100}
